async function parseXlsx(file){
  const buf=await file.arrayBuffer();
  const z=await unzipEntries(buf),dec=new TextDecoder('utf-8');
  const getXml=async name=>new DOMParser().parseFromString(dec.decode(await readZipEntry(z,name)),'application/xml');

    const xml=await getXml('xl/sharedStrings.xml');
    shared=[...xml.getElementsByTagName('si')].map(si=>[...si.getElementsByTagName('t')].map(t=>t.textContent||'').join(''));
  }
  let sheetName='xl/worksheets/sheet1.xml';
  if(!z.entries[sheetName]) sheetName=Object.keys(z.entries).find(k=>/^xl\/worksheets\/sheet\d+\.xml$/.test(k));
  if(!sheetName) throw new Error('Worksheet Excel tidak ditemukan');
  const xml=await getXml(sheetName);
  const xml=await getXml(sheetName);
  const rows=[...xml.getElementsByTagName('row')].map(r=>{
    const out=[];
    for(const c of [...r.children].filter(x=>x.localName==='c')){
      const ref=c.getAttribute('r')||'';
      const col=(ref.match(/^[A-Z]+/i)||['A'])[0].toUpperCase();
      let idx=0; for(const ch of col) idx=idx*26+ch.charCodeAt(0)-64; idx--;
      const t=c.getAttribute('t')||'';
      const v=c.getElementsByTagName('v')[0]?.textContent||'';
      let val=v;
      if(t==='s') val=shared[Number(v)]??'';
      else if(t==='inlineStr') val=[...c.getElementsByTagName('t')].map(x=>x.textContent||'').join('');
      else if(t==='b') val=v==='1'?'TRUE':'FALSE';
      else if(t==='str') val=c.getElementsByTagName('v')[0]?.textContent||'';
      out[idx]=val;
    }
    return out;
  });
  return rows.filter(r=>r.some(v=>String(v??'').trim()!==''));
}

}

function autoConnect(manual){
  if(localStorage.getItem('forceOffline')==='1'&&!manual)return;
  localStorage.removeItem('forceOffline');
  if(location.protocol==='file:'){online=false;net();if(manual)alert('Buka aplikasi melalui link Railway agar sinkronisasi online aktif.');return}
  // Always use the current application host. This prevents an old localStorage
  // server URL from silently connecting this device to an old server.
  serverUrl=(location.protocol==='https:'?'https://':'http://')+location.host;
  localStorage.setItem('syncServer',serverUrl);
  startWS(serverUrl.replace(/^http/,'ws'));
  startPolling();
}
function startPolling(){
  clearInterval(window._syncPoll);
  pollServer();
  window._syncPoll=setInterval(pollServer,2000);
}
async function pollServer(){
  if(localStorage.getItem('forceOffline')==='1'||location.protocol==='file:')return;
  try{
    const r=await fetch('/api/state',{cache:'no-store'}); if(!r.ok)throw Error('Server '+r.status);
    const m=await r.json(); if(!m.ok)throw Error(m.error||'Server error');
    const dirty=localStorage.getItem('syncDirty')==='1';
    const hasLocal=classes && Object.keys(classes).length>0;
    // On a brand-new PostgreSQL database, seed it from the first device before
    // accepting an empty server snapshot. This removes the race between the
    // initial HTTP poll and the WebSocket connection.
    if(Number(m.version)===0 && hasLocal){
      await syncSend(true);
      online=true;updateStatusEverywhere();return;
    }
    if(!dirty){
      applyServerState(m);
    }else if(active){
      // Keep the class currently being edited locally. Refresh other classes
      // silently; NEVER rebuild the visible page during background polling.
      const merged={...(m.classes||{}),[String(active)]:classes[String(active)]};
      classes=merged;
      localStorage.setItem('classData',JSON.stringify(classes));
      updateStatusEverywhere();
    }else{
      // Full-school dirty operations (Excel import/restore) are pushed explicitly.
      await syncSend(true);
    }
    online=true;updateStatusEverywhere();
  }catch(e){online=false;updateStatusEverywhere()}
}
function applyServerState(m){
  // IMPORTANT: synchronization must NEVER navigate or rebuild the current page.
  // Rebuilding app.innerHTML here used to make every click/form entry jump back
  // to Dashboard. The server state is updated silently; the user can navigate
  // normally and the next page render will use the newest state.
  if(m && m.classes && typeof m.classes==='object'){
    classes=m.classes;
    localStorage.setItem('classData',JSON.stringify(classes));
  }
  if(m && m.schoolSetting && typeof m.schoolSetting==='object'){
    school=m.schoolSetting;
    localStorage.setItem('schoolSetting',JSON.stringify(school));
  }
  localStorage.removeItem('syncDirty');
  lastLocalChange=0;
  localStorage.removeItem('lastLocalChange');
  updateStatusEverywhere();
}
function startWS(url){
  if(!url||location.protocol==='file:')return;
  if(ws&&[WebSocket.OPEN,WebSocket.CONNECTING].includes(ws.readyState))return;
  try{
    ws=new WebSocket(url);
    ws.onopen=()=>{online=true;net();const hello={type:'join',school:SCHOOL_CODE,classes:classes,dirty:localStorage.getItem('syncDirty')==='1',changedAt:Number(lastLocalChange)||0,clientId:getClientId()};if(adminToken)hello.schoolSetting=school;ws.send(JSON.stringify(hello));};
    ws.onmessage=e=>{try{const m=JSON.parse(e.data);if(m.type==='sync'&&m.classes&&!localStorage.getItem('syncDirty'))applyServerState(m)}catch(err){console.error(err)}};
    ws.onerror=()=>{online=false;net()};
    ws.onclose=()=>{online=false;net();clearTimeout(retryTimer);if(localStorage.getItem('forceOffline')!=='1')retryTimer=setTimeout(()=>startWS(url),3000)};
  }catch{online=false;net()}
}

async function syncSend(force=false){
  if(location.protocol==='file:'||localStorage.getItem('forceOffline')==='1')return false;
  // Class/teacher data sync must never be blocked by an old schoolSetting
  // cached on a teacher device. Protected identity settings are sent only by
  // an authenticated Admin when explicitly forcing a full save.
  const payload={school:SCHOOL_CODE,changedAt:lastLocalChange,clientId:getClientId()};
  if(force||!active){payload.mode='full';payload.classes=classes;}
  else {payload.mode='class';payload.classId=String(active);payload.classData=classes[String(active)]||{};}
  if(force && adminToken) payload.schoolSetting=school;
  try{
    const r=await fetch('/api/state',{method:'POST',headers:adminHeaders(),body:JSON.stringify(payload),cache:'no-store'});
    if(!r.ok)throw Error('Server '+r.status);
    const m=await r.json();if(!m.ok)throw Error(m.error||'Gagal sinkron');
    // Do not call any navigation/render function here. For a class save, merge
    // the acknowledged class into local state; for an Admin full save, accept
    // the complete server snapshot.
    if(m.classes && typeof m.classes==='object'){
      if(payload.mode==='class' && active){
        classes={...(m.classes||{}),[String(active)]:classes[String(active)]};
      }else{
        classes=m.classes;
      }
      localStorage.setItem('classData',JSON.stringify(classes));
    }
    if(payload.mode==='full' && m.schoolSetting && adminToken){
      school=m.schoolSetting;
      localStorage.setItem('schoolSetting',JSON.stringify(school));
    }
    localStorage.removeItem('syncDirty');
    lastLocalChange=0; localStorage.removeItem('lastLocalChange');
    online=true;updateStatusEverywhere();return true;
  }catch(e){online=false;updateStatusEverywhere();return false}
}
